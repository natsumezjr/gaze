# 眼动追踪校准系统前后端集成接口文档

## 概述

本文档详细说明了眼动追踪校准前端界面与后端Python系统的集成方案，基于项目现有的`project.fitting`和`project.recognition`模块接口，提供完整的前后端数据交换和功能集成方案。

---

## 📁 项目结构

```
pzs/
├── eye_tracking_calibration.html    # 前端校准界面（单文件集成）
├── backend_structure_api.md         # 后端架构与接口文档
├── 前后端集成接口文档.md            # 本文档
└── 其他技术文档...
```

**前端特性：**
- 单文件HTML集成（HTML + CSS + JavaScript）
- 支持矩形和六边形两种校准点布局
- 三种背景主题（白色、黑色、浅色）
- 实时视线检测和暂停功能
- 中心点特殊标识
- 进度显示和状态管理

## 🔄 数据流架构

### 整体流程
```
前端校准界面 → 数据收集 → JSON格式输出 → 后端处理 → Kappa校准 → 结果返回
```

### 核心模块对应
- **前端校准界面** → 提供校准点位、时序控制和数据收集
- **后端API接口** → 基于`project.fitting.app.api`的会话管理和数据处理
- **数据管理模块** → `project.recg_fit_data.data_manager`进行坐标管理
- **拟合模块** → `project.fitting.core`进行眼球拟合和Kappa校准

### 2. 集成架构设计

#### 桥接应用集成方案
```python
# calibration_bridge.py - 前后端桥接应用
import cv2
import numpy as np
import json
import time
from datetime import datetime
from typing import Dict, List, Optional

# 导入现有模块
from project.fitting.app.api import start_session, push_sample, fit_kappa, compute_gaze
from project.fitting.app.state import CalibrationSample, KappaModel, GazeVector
from project.recg_fit_data.data_manager import RecgFitDataManager

class CalibrationBridge:
    def __init__(self):
        # 初始化数据管理器
        self.data_manager = RecgFitDataManager()
        
        # 校准状态
        self.session_id = None
        self.is_collecting = False
        self.current_point_data = []
        self.all_calibration_data = []
        
    def start_calibration_session(self, session_id: str = None) -> Dict:
        """开始校准会话"""
        if not session_id:
            session_id = f"calibration_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        
        self.session_id = session_id
        self.all_calibration_data = []
        
        # 启动后端会话
        result = start_session(session_id)
        
        print(f"校准会话开始: {session_id}")
        return result
        
    def process_calibration_data(self, json_data: Dict) -> Dict:
        """
        处理前端返回的校准数据
        
        Args:
            json_data: 前端校准数据JSON
            
        Returns:
            Dict: 处理结果
        """
        try:
            # 1. 验证JSON数据格式
            self._validate_calibration_data(json_data)
            
            # 2. 提取数据
            eyes = np.array(json_data["eyes"])
            pupils = np.array(json_data["pupils"])
            target_pixels = json_data["target_pixels"]
            camera_matrix = np.array(json_data["camera_matrix"])
            screen_resolution = json_data["screen_resolution"]
            
            # 3. 构建校准样本
            samples = self._convert_json_to_samples(eyes, pupils, target_pixels, camera_matrix)
            
            # 4. 推送样本到后端
            for sample in samples:
                payload = {
                    "timestamp": sample.timestamp,
                    "target_pixel": sample.target_pixel,
                    "eye": "left",  # 默认左眼，可根据需要调整
                    "frame_id": sample.frame_id
                }
                push_sample(payload)
            
            # 5. 执行Kappa校准
            kappa_result = fit_kappa(self.session_id, camera_matrix)
            
            return {
                "success": True,
                "message": "校准数据处理成功",
                "processed_samples": len(samples),
                "kappa_model": kappa_result.get("kappa_model"),
                "session_id": self.session_id
            }
            
        except Exception as e:
            return {
                "success": False,
                "message": f"校准数据处理失败: {str(e)}",
                "error_code": "PROCESSING_FAILED"
            }
    
    def _validate_calibration_data(self, json_data: Dict) -> None:
        """验证前端JSON数据格式"""
        required_fields = ["metadata", "eyes", "pupils", "target_pixels", "camera_matrix", "screen_resolution"]
        
        for field in required_fields:
            if field not in json_data:
                raise ValueError(f"缺少必需字段: {field}")
        
        # 验证数组长度一致性
        eyes = json_data["eyes"]
        pupils = json_data["pupils"]
        target_pixels = json_data["target_pixels"]
        
        if not (len(eyes) == len(pupils) == len(target_pixels)):
            raise ValueError("eyes、pupils、target_pixels数组长度不一致")
        
        # 验证相机矩阵
        camera_matrix = json_data["camera_matrix"]
        if len(camera_matrix) != 3 or any(len(row) != 3 for row in camera_matrix):
            raise ValueError("camera_matrix必须是3x3矩阵")
    
    def _convert_json_to_samples(self, eyes: np.ndarray, pupils: np.ndarray, 
                                target_pixels: List, camera_matrix: np.ndarray) -> List[CalibrationSample]:
        """将JSON数据转换为校准样本列表"""
        samples = []
        
        for i in range(len(eyes)):
            # 计算理论视线向量
            theoretical_gaze = self._compute_theoretical_gaze(eyes[i], pupils[i], target_pixels[i], camera_matrix)
            
            sample = CalibrationSample(
                timestamp=time.time(),
                target_pixel=tuple(target_pixels[i]),
                eye_center=eyes[i],
                pupil_center=pupils[i],
                theoretical_gaze=theoretical_gaze,
                quality=1.0,  # 默认质量分数
                frame_id=i
            )
            samples.append(sample)
        
        return samples
    
    def _compute_theoretical_gaze(self, eye_center: np.ndarray, pupil_center: np.ndarray, 
                                 target_pixel: List, camera_matrix: np.ndarray) -> np.ndarray:
        """计算理论视线向量"""
        # 将屏幕像素坐标转换为相机坐标系
        # 这里需要根据具体的相机标定参数进行转换
        # 简化实现，实际使用时需要完善
        target_3d = np.array([target_pixel[0], target_pixel[1], 1.0])
        
        # 计算从眼球中心到目标的向量
        gaze_vector = target_3d - eye_center
        gaze_vector = gaze_vector / np.linalg.norm(gaze_vector)  # 归一化
        
        return gaze_vector
    
    def get_calibration_result(self) -> Dict:
        """获取校准结果"""
        if not self.session_id:
            return {"error": "没有活跃的校准会话"}
        
        try:
            # 获取Kappa模型
            kappa_result = fit_kappa(self.session_id)
            
            # 计算视线
            gaze_result = compute_gaze("left")
            
            return {
                "session_id": self.session_id,
                "kappa_model": kappa_result.get("kappa_model"),
                "gaze_vector": gaze_result.get("gaze"),
                "status": "completed"
            }
            
        except Exception as e:
            return {
                "error": f"获取校准结果失败: {str(e)}",
                "status": "failed"
            }

# 使用示例
if __name__ == "__main__":
    bridge = CalibrationBridge()
    
    # 开始校准会话
    session_result = bridge.start_calibration_session()
    print("会话启动结果:", session_result)
```

### 推荐的简化集成方案

#### 直接HTTP API方案（最简单）
```python
# 创建一个简单的HTTP服务器包装现有模块
from http.server import HTTPServer, BaseHTTPRequestHandler
import json
import threading

class CalibrationHandler(BaseHTTPRequestHandler):
    bridge = CalibrationBridge()  # 全局桥接对象
    
    def do_POST(self):
        if self.path == '/process_calibration':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            json_data = json.loads(post_data.decode('utf-8'))
            
            # 处理校准数据
            result = self.bridge.process_calibration_data(json_data)
            
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps(result).encode())
            
        elif self.path == '/start_session':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            data = json.loads(post_data.decode('utf-8'))
            
            session_id = data.get('session_id')
            result = self.bridge.start_calibration_session(session_id)
            
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps(result).encode())
            
        elif self.path == '/get_result':
            result = self.bridge.get_calibration_result()
            
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps(result).encode())
    
    def do_OPTIONS(self):
        # 处理CORS预检请求
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'POST, GET, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()

# 启动服务器
if __name__ == "__main__":
    httpd = HTTPServer(('localhost', 8000), CalibrationHandler)
    print("校准桥接服务器启动在 http://localhost:8000")
    httpd.serve_forever()
```
