### 2. 集成架构设计

#### 桌面应用集成方案
```python
# calibration_bridge.py - 前后端桥接应用
import cv2
import numpy as np
import json
import threading
import time
from datetime import datetime

# 导入现有模块
from project.recognition.core.detector import FaceDetector
from project.recognition.utils.camera_calibration import CameraCalibrator
from project.recognition.utils.camera_data_manager import add_frame, get_image, get_depth
from project.fitting.core.kappa_calibrator import estimate_kappa

class CalibrationBridge:
    def __init__(self):
        # 初始化现有组件
        self.calibrator = CameraCalibrator(rgb_d=True)
        self.camera_params = self.calibrator.load_camera_params()
        self.cap = self.calibrator.get_cap()
        self.face_detector = FaceDetector(self.camera_params, rgb_d=True)
        
        # 校准状态
        self.is_collecting = False
        self.current_point_data = []
        self.all_calibration_data = []
        
    def start_calibration_session(self):
        """开始校准会话"""
        self.all_calibration_data = []
        print("校准会话开始")
        
    def start_point_collection(self, point_index, screen_xy):
        """开始采集指定校准点的数据"""
        self.is_collecting = True
        self.current_point_data = []
        self.current_point = {"index": point_index, "screen_xy": screen_xy}
        print(f"开始采集校准点 {point_index + 1}: {screen_xy}")
        
    def collect_frame_data(self):
        """采集当前帧的眼动数据"""
        if not self.is_collecting:
            return None
            
        ret, frame = self.cap.read()
        if not ret:
            return None
            
        # 使用固定深度图（基于main.py的实现）
        h, w = frame.shape[:2]
        fixed_depth_meters = 0.6
        depth_map = np.ones((h, w), dtype=np.float32) * (fixed_depth_meters / self.camera_params['depth_scale'])
        
        # 检测人脸
        if self.face_detector.detect_face(frame, depth_map):
            # 获取检测结果（使用core.md中定义的接口）
            pupil_center = self.face_detector.get_pupil_center()
            confidence = self.face_detector.get_detection_confidence()
            
            # 构建样本数据
            sample_data = {
                "timestamp": datetime.now().isoformat(),
                "pupil_center": {
                    "left": pupil_center["left"].tolist(),
                    "right": pupil_center["right"].tolist()
                },
                "confidence": float(confidence),
                "screen_target": self.current_point["screen_xy"]
            }
            
            self.current_point_data.append(sample_data)
            return sample_data
        
        return None
        
    def stop_point_collection(self):
        """停止当前点的数据采集"""
        self.is_collecting = False
        
        # 将当前点数据添加到总数据中
        point_result = {
            "point_index": self.current_point["index"],
            "screen_xy": self.current_point["screen_xy"],
            "samples": self.current_point_data,
            "sample_count": len(self.current_point_data)
        }
        
        self.all_calibration_data.append(point_result)
        print(f"完成校准点 {self.current_point['index'] + 1}，采集样本数: {len(self.current_point_data)}")
        
        return point_result
        
    def complete_calibration(self):
        """完成校准，进行Kappa标定"""
        try:
            # 转换数据格式供fitting模块使用
            samples = self._convert_to_fitting_format()
            
            # 调用fitting模块进行Kappa估计
            kappa_params, info = estimate_kappa(samples)
            
            # 计算校准精度
            accuracy = info.get("mean_ang_err_deg", 0)
            accuracy_percent = max(0, min(100, 100 - accuracy * 10))  # 转换为百分比
            
            result = {
                "status": "success",
                "model_id": f"calibration_{datetime.now().strftime('%Y%m%d_%H%M%S')}",
                "accuracy": accuracy_percent,
                "kappa_params": {
                    "kappa_rad": kappa_params.tolist(),
                    "kappa_deg": np.degrees(kappa_params).tolist(),
                    "mean_error_deg": info.get("mean_ang_err_deg", 0),
                    "median_error_deg": info.get("median_ang_err_deg", 0)
                },
                "samples_processed": info.get("num_samples", 0),
                "calibration_info": info
            }
            
            print("校准完成:", result)
            return result
            
        except Exception as e:
            error_result = {
                "status": "error",
                "error_code": "CALIBRATION_FAILED",
                "message": f"校准失败: {str(e)}"
            }
            print("校准失败:", error_result)
            return error_result
    
    def _convert_to_fitting_format(self):
        """将采集的数据转换为fitting模块需要的格式"""
        # 根据您的fitting模块接口实现数据转换
        # 这需要参考具体的Sample类定义
        samples = []
        
        for point_data in self.all_calibration_data:
            screen_xy = point_data["screen_xy"]
            
            for sample in point_data["samples"]:
                # 构建fitting模块需要的Sample对象
                # 具体实现需要参考您的fitting模块接口
                sample_obj = self._create_sample_object(screen_xy, sample)
                samples.append(sample_obj)
                
        return samples
    
    def _create_sample_object(self, screen_xy, sample_data):
        """创建fitting模块需要的Sample对象"""
        # 这里需要根据您的fitting模块的具体Sample类实现
        # 暂时返回字典格式，实际使用时需要调整
        return {
            "screen_target": screen_xy,
            "pupil_center": sample_data["pupil_center"],
            "confidence": sample_data["confidence"],
            "timestamp": sample_data["timestamp"]
        }

# 使用示例
if __name__ == "__main__":
    bridge = CalibrationBridge()
    
    # 开始校准会话
    bridge.start_calibration_session()
    
    # 校准第一个点
    bridge.start_point_collection(0, [33, 20])
    
    # 采集数据（持续2秒）
    start_time = time.time()
    while time.time() - start_time < 2.0:
        bridge.collect_frame_data()
        time.sleep(0.1)  # 100ms间隔
    
    # 停止采集
    bridge.stop_point_collection()
    
    # ... 重复其他校准点
    
    # 完成校准
    result = bridge.complete_calibration()
```

```python
# Python桌面应用：监听指令文件
import json
import os
import time
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class CalibrationCommandHandler(FileSystemEventHandler):
    def __init__(self, bridge):
        self.bridge = bridge
        
    def on_modified(self, event):
        if event.src_path.endswith('calibration_command.json'):
            with open(event.src_path, 'r') as f:
                command = json.load(f)
                
            if command['action'] == 'start_point':
                self.bridge.start_point_collection(
                    command['point_index'], 
                    command['screen_xy']
                )
            elif command['action'] == 'stop_point':
                result = self.bridge.stop_point_collection()
                # 写入结果文件供前端读取
                with open('calibration_result.json', 'w') as f:
                    json.dump(result, f)
```

```python
# Python桌面应用：WebSocket服务器
import asyncio
import websockets
import json

class CalibrationWebSocketServer:
    def __init__(self, bridge):
        self.bridge = bridge
        
    async def handle_client(self, websocket, path):
        async for message in websocket:
            command = json.loads(message)
            
            if command['action'] == 'start_point':
                self.bridge.start_point_collection(
                    command['point_index'],
                    command['screen_xy']
                )
            elif command['action'] == 'stop_point':
                result = self.bridge.stop_point_collection()
                await websocket.send(json.dumps({
                    "action": "point_complete",
                    "data": result
                }))

# 启动WebSocket服务器
start_server = websockets.serve(
    CalibrationWebSocketServer(bridge).handle_client,
```

### 推荐的简化集成方案

#### 直接调用方案（最简单）
```python
# 创建一个简单的HTTP服务器包装现有模块
from http.server import HTTPServer, BaseHTTPRequestHandler
import json

class CalibrationHandler(BaseHTTPRequestHandler):
    bridge = CalibrationBridge()  # 全局桥接对象
    
    def do_POST(self):
        if self.path == '/start_point':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            command = json.loads(post_data.decode('utf-8'))
            
            self.bridge.start_point_collection(
                command['point_index'],
                command['screen_xy']
            )
            
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps({"status": "started"}).encode())
            
        elif self.path == '/stop_point':
            result = self.bridge.stop_point_collection()
            
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps(result).encode())

# 启动服务器
httpd = HTTPServer(('localhost', 8000), CalibrationHandler)
httpd.serve_forever()
```
